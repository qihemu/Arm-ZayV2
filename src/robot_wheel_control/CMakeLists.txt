cmake_minimum_required(VERSION 3.16)
project(robot_wheel_control VERSION 0.2.0 LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(damiao_core CONFIG REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(robot_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)

# ROS无关的配置、收发线程及相对运动闭环运行层。
add_library(wheel_runtime SHARED src/wheel_configuration.cpp src/wheel_runtime.cpp src/wheel_relative_motion.cpp)
target_compile_features(wheel_runtime PUBLIC cxx_std_17)
target_include_directories(wheel_runtime PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(wheel_runtime damiao_core::damiao_core yaml-cpp Threads::Threads)
target_compile_options(wheel_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_library(wheel_ros_api SHARED src/wheel_ros_api.cpp)
target_link_libraries(wheel_ros_api wheel_runtime)
ament_target_dependencies(wheel_ros_api rclcpp robot_interfaces geometry_msgs sensor_msgs)
add_library(h55_base_hardware SHARED src/h55_base_hardware.cpp)
target_link_libraries(h55_base_hardware wheel_ros_api)
ament_target_dependencies(h55_base_hardware hardware_interface pluginlib rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(hardware_interface wheel_plugins.xml)
add_executable(wheel_bench_node src/wheel_bench_node.cpp)
target_link_libraries(wheel_bench_node wheel_ros_api)
ament_target_dependencies(wheel_bench_node rclcpp)
install(TARGETS wheel_runtime wheel_ros_api h55_base_hardware LIBRARY DESTINATION lib)
add_executable(wheel_readback src/wheel_readback.cpp)
target_link_libraries(wheel_readback wheel_runtime)
install(TARGETS wheel_bench_node wheel_readback DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES ../../config/robot_wheel_control.yaml DESTINATION share/${PROJECT_NAME}/config)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY launch docs DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/wheel_cli.py DESTINATION lib/${PROJECT_NAME} RENAME wheel_cli)
ament_package()
